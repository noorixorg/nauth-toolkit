# Quick Start

Install nauth-toolkit and add authentication to your Node.js application in minutes.

## Installation

:::info Yarn 1.x Users
If you're using **Yarn 1.x (Classic)**, you'll need to manually install peer dependencies. Yarn 1.x doesn't automatically install peer dependencies.

**npm, pnpm, bun, and Yarn 2+** all handle peer dependencies automatically, so no manual installation is needed.

See the [Peer Dependencies](#peer-dependencies) section below for the complete list if you're using Yarn 1.x.
:::

## Basic Setup

<Tabs groupId="framework">
<TabItem value="nestjs" label="NestJS" default>

```typescript title="main.ts"
import { NestFactory } from '@nestjs/core';
import { ValidationPipe } from '@nestjs/common';
import * as dotenv from 'dotenv';
import { NAuthHttpExceptionFilter } from '@nauth-toolkit/nestjs';

// Load environment variables BEFORE importing AppModule
dotenv.config();

import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  // Enable global validation pipe for DTO validation
  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,
      forbidNonWhitelisted: true,
      transform: true,
    }),
  );

  // Enable nauth-toolkit exception filter
  app.useGlobalFilters(new NAuthHttpExceptionFilter());

  await app.listen(3000);
}
bootstrap();
```

```typescript title="app.module.ts"
import { Module } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';
import { AuthModule } from '@nauth-toolkit/nestjs';
import { getNAuthEntities, getNAuthTransientStorageEntities } from '@nauth-toolkit/database-typeorm-postgres';
import { authConfig } from './config/auth.config';

@Module({
  imports: [
    TypeOrmModule.forRoot({
      type: 'postgres',
      host: process.env.DB_HOST,
      port: parseInt(process.env.DB_PORT || '5432'),
      username: process.env.DB_USERNAME,
      password: process.env.DB_PASSWORD,
      database: process.env.DB_DATABASE,
      // Storage entities are required when using DatabaseStorageAdapter.
      // If you use RedisStorageAdapter instead, you can omit getNAuthTransientStorageEntities().
      entities: [...getNAuthEntities(), ...getNAuthTransientStorageEntities()],
    }),
    AuthModule.forRoot(authConfig),
  ],
})
export class AppModule {}
```

```typescript title="auth.controller.ts"
import { Controller, Post, Get, Body, UseGuards } from '@nestjs/common';
import { AuthService, SignupDTO, LoginDTO, AuthGuard, CurrentUser, Public, IUser } from '@nauth-toolkit/nestjs';

@Controller('auth')
export class AuthController {
  constructor(private readonly authService: AuthService) {}

  @Public()
  @Post('signup')
  async signup(@Body() dto: SignupDTO) {
    return this.authService.signup(dto);
  }

  @Public()
  @Post('login')
  async login(@Body() dto: LoginDTO) {
    return this.authService.login(dto);
  }

  @UseGuards(AuthGuard)
  @Get('profile')
  async getProfile(@CurrentUser() user: IUser) {
    return user;
  }
}
```

<TabItem value="express" label="Express">

```typescript title="index.ts"
import express from 'express';
import cookieParser from 'cookie-parser';
import { DataSource } from 'typeorm';
import { NAuth, ExpressAdapter, NAuthInstance, ExpressMiddlewareType } from '@nauth-toolkit/core';
import { getNAuthEntities, getNAuthTransientStorageEntities } from '@nauth-toolkit/database-typeorm-postgres';
import { authConfig } from './config/auth.config';
import { createAuthRoutes } from './routes/auth.routes';

async function bootstrap() {
  // Initialize database
  const dataSource = new DataSource({
    type: 'postgres',
    host: process.env.DB_HOST,
    port: parseInt(process.env.DB_PORT || '5432'),
    username: process.env.DB_USERNAME,
    password: process.env.DB_PASSWORD,
    database: process.env.DB_DATABASE,
    // Storage entities are required when using DatabaseStorageAdapter.
    // If you use RedisStorageAdapter instead, you can omit getNAuthTransientStorageEntities().
    entities: [...getNAuthEntities(), ...getNAuthTransientStorageEntities()],
    synchronize: process.env.NODE_ENV === 'development',
  });
  await dataSource.initialize();

  // Create NAuth instance
  const nauth = await NAuth.create({
    config: authConfig,
    dataSource,
    adapter: new ExpressAdapter(),
  });

  const app = express();
  app.use(express.json());
  app.use(cookieParser());

  // Mount middleware (order matters)
  app.use(nauth.middleware.clientInfo); // FIRST - initializes context
  app.use(nauth.middleware.csrf);
  app.use(nauth.middleware.auth);
  app.use(nauth.middleware.tokenDelivery);

  // Routes
  app.use('/auth', createAuthRoutes(nauth));

  app.listen(3000, () => console.log('Server running on :3000'));
}

bootstrap();
```

```typescript title="routes/auth.routes.ts"
import { Router, Request, Response, NextFunction } from 'express';
import { NAuthInstance, ExpressMiddlewareType, RequestHandler } from '@nauth-toolkit/core';

export function createAuthRoutes(nauth: NAuthInstance<ExpressMiddlewareType, RequestHandler>): Router {
  const router = Router();

  router.post('/signup', nauth.helpers.public(), async (req: Request, res: Response, next: NextFunction) => {
    try {
      const result = await nauth.authService.signup(req.body);
      res.status(201).json(result);
    } catch (error) {
      next(error);
    }
  });

  router.post('/login', nauth.helpers.public(), async (req: Request, res: Response, next: NextFunction) => {
    try {
      const result = await nauth.authService.login(req.body);
      res.json(result);
    } catch (error) {
      next(error);
    }
  });

  router.get('/me', nauth.helpers.requireAuth(), (req: Request, res: Response) => {
    const user = nauth.helpers.getCurrentUser();
    res.json(user);
  });

  return router;
}
```

```typescript title="index.ts"
import Fastify, { FastifyInstance } from 'fastify';
import cookie from '@fastify/cookie';
import { DataSource } from 'typeorm';
import { NAuth, FastifyAdapter, NAuthInstance } from '@nauth-toolkit/core';
import { getNAuthEntities, getNAuthTransientStorageEntities } from '@nauth-toolkit/database-typeorm-postgres';
import { authConfig } from './config/auth.config';
import { createAuthRoutes } from './routes/auth.routes';

async function bootstrap() {
  // Initialize database
  const dataSource = new DataSource({
    type: 'postgres',
    host: process.env.DB_HOST,
    port: parseInt(process.env.DB_PORT || '5432'),
    username: process.env.DB_USERNAME,
    password: process.env.DB_PASSWORD,
    database: process.env.DB_DATABASE,
    // Storage entities are required when using DatabaseStorageAdapter.
    // If you use RedisStorageAdapter instead, you can omit getNAuthTransientStorageEntities().
    entities: [...getNAuthEntities(), ...getNAuthTransientStorageEntities()],
    synchronize: process.env.NODE_ENV === 'development',
  });
  await dataSource.initialize();

  // Create NAuth instance
  const nauth = await NAuth.create({
    config: authConfig,
    dataSource,
    adapter: new FastifyAdapter(),
  });

  const fastify = Fastify();
  await fastify.register(cookie);

  // Register hooks (order matters)
  fastify.addHook('onRequest', nauth.middleware.clientInfo as any); // FIRST
  fastify.addHook('onRequest', nauth.middleware.csrf as any);
  fastify.addHook('onRequest', nauth.middleware.auth as any);
  fastify.addHook('onSend', nauth.middleware.tokenDelivery as any);

  // Routes
  await fastify.register(async (api) => {
    createAuthRoutes(api, nauth);
  }, { prefix: '/auth' });

  await fastify.listen({ port: 3000 });
  console.log('Server running on :3000');
}

bootstrap();
```

```typescript title="routes/auth.routes.ts"
import { FastifyInstance } from 'fastify';
import { NAuthInstance } from '@nauth-toolkit/core';

export async function createAuthRoutes(
  fastify: FastifyInstance,
  nauth: NAuthInstance<any, any>,
): Promise<void> {
  fastify.post(
    '/signup',
    { preHandler: nauth.helpers.public() as any },
    nauth.adapter.wrapRouteHandler(async (req) => {
      return nauth.authService.signup(req.body as any);
    }),
  );

  fastify.post(
    '/login',
    { preHandler: nauth.helpers.public() as any },
    nauth.adapter.wrapRouteHandler(async (req) => {
      return nauth.authService.login(req.body as any);
    }),
  );

  fastify.get(
    '/me',
    { preHandler: nauth.helpers.requireAuth() as any },
    nauth.adapter.wrapRouteHandler(async () => nauth.helpers.getCurrentUser()),
  );
}
```

## Key Differences by Framework

| Aspect | NestJS | Express | Fastify |
|--------|--------|---------|---------|
| **Bootstrap** | `AuthModule.forRoot()` | `NAuth.create()` | `NAuth.create()` |
| **Adapter** | Automatic | `ExpressAdapter` | `FastifyAdapter` |
| **Route protection** | `@UseGuards(AuthGuard)` | `nauth.helpers.requireAuth()` | `preHandler` + `nauth.helpers.requireAuth()` |
| **Public routes** | `@Public()` decorator | `nauth.helpers.public()` | `nauth.helpers.public()` |
| **Get current user** | `@CurrentUser()` decorator | `nauth.helpers.getCurrentUser()` | `nauth.helpers.getCurrentUser()` |
| **Context propagation** | Automatic | Automatic | Use `adapter.wrapRouteHandler` on route handlers |

## Troubleshooting

### TypeScript IntelliSense Not Working (Express/Fastify)

If you're not getting IntelliSense/autocomplete for the `nauth` instance in Express or Fastify, you need to explicitly type it:

<Tabs>
<TabItem value="express-typing" label="Express">

```typescript
import { NAuthInstance, ExpressMiddlewareType, RequestHandler } from '@nauth-toolkit/core';

// Type the nauth instance explicitly
let nauth: NAuthInstance<ExpressMiddlewareType, RequestHandler>;

async function bootstrap() {
  nauth = await NAuth.create({
    config: authConfig,
    dataSource,
    adapter: new ExpressAdapter(),
  });

  // Now you'll get full IntelliSense for:
  // nauth.authService.*
  // nauth.middleware.*
  // nauth.helpers.*
  // etc.
}
```

Or pass it as a typed parameter to your route functions:

```typescript
export function createAuthRoutes(
  nauth: NAuthInstance<ExpressMiddlewareType, RequestHandler>
): Router {
  // Full IntelliSense available here
  const router = Router();
  // ...
}
```

</TabItem>
<TabItem value="fastify-typing" label="Fastify">

```typescript
import { NAuthInstance, FastifyMiddlewareType, FastifyHookHandler } from '@nauth-toolkit/core';

// Type the nauth instance explicitly
let nauth: NAuthInstance<FastifyMiddlewareType, FastifyHookHandler>;

async function bootstrap() {
  nauth = await NAuth.create({
    config: authConfig,
    dataSource,
    adapter: new FastifyAdapter(),
  });

  // Now you'll get full IntelliSense
}
```

Or in your route functions:

```typescript
export async function createAuthRoutes(
  fastify: FastifyInstance,
  nauth: NAuthInstance<FastifyMiddlewareType, FastifyHookHandler>,
): Promise<void> {
  // Full IntelliSense available here
}
```

</TabItem>
</Tabs>

:::info Why is this needed?
TypeScript's type inference for `NAuth.create()` is limited because the return type depends on which adapter you provide. Explicitly typing the variable helps TypeScript understand the exact shape of your `nauth` instance.
:::
