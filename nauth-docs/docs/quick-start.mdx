# Quick Start

Install nauth-toolkit and add authentication to your Node.js application in minutes.

## Installation

:::info Yarn 1.x Users
If you're using **Yarn 1.x (Classic)**, you'll need to manually install peer dependencies. Yarn 1.x doesn't automatically install peer dependencies.

**npm, pnpm, bun, and Yarn 2+** all handle peer dependencies automatically, so no manual installation is needed.

See the [Peer Dependencies](#peer-dependencies) section below for the complete list if you're using Yarn 1.x.
:::

## Basic Setup

```typescript title="main.ts"
import { NestFactory } from '@nestjs/core';
import { ValidationPipe } from '@nestjs/common';
import * as dotenv from 'dotenv';
import { NAuthHttpExceptionFilter } from '@nauth-toolkit/nestjs';

// Load environment variables BEFORE importing AppModule
dotenv.config();

import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  // Enable global validation pipe for DTO validation
  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,
      forbidNonWhitelisted: true,
      transform: true,
    }),
  );

  // Enable nauth-toolkit exception filter
  app.useGlobalFilters(new NAuthHttpExceptionFilter());

  await app.listen(3000);
}
bootstrap();
```

```typescript title="app.module.ts"
import { Module } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';
import { AuthModule } from '@nauth-toolkit/nestjs';
import {
  getNAuthEntities,
  getNAuthTransientStorageEntities,
} from '@nauth-toolkit/database-typeorm-postgres';
import { authConfig } from './config/auth.config';
import { AuthController } from './auth.controller';
import { AppController } from './app.controller';
import { AppService } from './app.service';

@Module({
  imports: [
    TypeOrmModule.forRoot({
      type: 'postgres',
      host: process.env.DB_HOST,
      port: parseInt(process.env.DB_PORT || '5432'),
      username: process.env.DB_USERNAME,
      password: process.env.DB_PASSWORD,
      database: process.env.DB_DATABASE,
      // Storage entities are required when using DatabaseStorageAdapter.
      // If you use RedisStorageAdapter instead, you can omit getNAuthTransientStorageEntities().
      entities: [...getNAuthEntities(), ...getNAuthTransientStorageEntities()],
    }),
    AuthModule.forRoot(authConfig),
  ],
  controllers: [AppController, AuthController],
  providers: [AppService],
})
export class AppModule {}
```

```typescript title="auth.controller.ts"
import { Controller, Post, Get, Body, UseGuards } from '@nestjs/common';
import { AuthService, SignupDTO, LoginDTO, AuthGuard, CurrentUser, Public } from '@nauth-toolkit/nestjs';
import type { IUser } from '@nauth-toolkit/nestjs';

@Controller('auth')
export class AuthController {
  constructor(private readonly authService: AuthService) {}

  @Public()
  @Post('signup')
  async signup(@Body() dto: SignupDTO) {
    return this.authService.signup(dto);
  }

  @Public()
  @Post('login')
  async login(@Body() dto: LoginDTO) {
    return this.authService.login(dto);
  }

  @UseGuards(AuthGuard)
  @Get('profile')
  getProfile(@CurrentUser() user: IUser) {
    return user;
  }
}
```

```typescript title="index.ts"
import express from 'express';
import cookieParser from 'cookie-parser';
import { DataSource } from 'typeorm';
import { NAuth, ExpressAdapter, NAuthInstance, ExpressMiddlewareType } from '@nauth-toolkit/core';
import { getNAuthEntities, getNAuthTransientStorageEntities } from '@nauth-toolkit/database-typeorm-postgres';
import { authConfig } from './config/auth.config';
import { createAuthRoutes } from './routes/auth.routes';

async function bootstrap() {
  // Initialize database
  const dataSource = new DataSource({
    type: 'postgres',
    host: process.env.DB_HOST,
    port: parseInt(process.env.DB_PORT || '5432'),
    username: process.env.DB_USERNAME,
    password: process.env.DB_PASSWORD,
    database: process.env.DB_DATABASE,
    // Storage entities are required when using DatabaseStorageAdapter.
    // If you use RedisStorageAdapter instead, you can omit getNAuthTransientStorageEntities().
    entities: [...getNAuthEntities(), ...getNAuthTransientStorageEntities()],
    synchronize: process.env.NODE_ENV === 'development',
  });
  await dataSource.initialize();

  // Create NAuth instance
  const nauth = await NAuth.create({
    config: authConfig,
    dataSource,
    adapter: new ExpressAdapter(),
  });

  const app = express();
  app.use(express.json());
  app.use(cookieParser());

  // Mount middleware (order matters)
  app.use(nauth.middleware.clientInfo); // FIRST - initializes context
  app.use(nauth.middleware.csrf);
  app.use(nauth.middleware.auth);
  app.use(nauth.middleware.tokenDelivery);

  // Routes
  app.use('/auth', createAuthRoutes(nauth));

  app.listen(3000, () => console.log('Server running on :3000'));
}

bootstrap();
```

```typescript title="routes/auth.routes.ts"
import { Router, Request, Response, NextFunction } from 'express';
import { NAuthInstance, ExpressMiddlewareType, RequestHandler } from '@nauth-toolkit/core';

export function createAuthRoutes(nauth: NAuthInstance<ExpressMiddlewareType, RequestHandler>): Router {
  const router = Router();

  router.post('/signup', nauth.helpers.public(), async (req: Request, res: Response, next: NextFunction) => {
    try {
      const result = await nauth.authService.signup(req.body);
      res.status(201).json(result);
    } catch (error) {
      next(error);
    }
  });

  router.post('/login', nauth.helpers.public(), async (req: Request, res: Response, next: NextFunction) => {
    try {
      const result = await nauth.authService.login(req.body);
      res.json(result);
    } catch (error) {
      next(error);
    }
  });

  router.get('/me', nauth.helpers.requireAuth(), (req: Request, res: Response) => {
    const user = nauth.helpers.getCurrentUser();
    res.json(user);
  });

  return router;
}
```

```typescript title="index.ts"
import Fastify, { FastifyInstance } from 'fastify';
import cookie from '@fastify/cookie';
import { DataSource } from 'typeorm';
import { NAuth, FastifyAdapter, NAuthInstance } from '@nauth-toolkit/core';
import { getNAuthEntities, getNAuthTransientStorageEntities } from '@nauth-toolkit/database-typeorm-postgres';
import { authConfig } from './config/auth.config';
import { createAuthRoutes } from './routes/auth.routes';

async function bootstrap() {
  // Initialize database
  const dataSource = new DataSource({
    type: 'postgres',
    host: process.env.DB_HOST,
    port: parseInt(process.env.DB_PORT || '5432'),
    username: process.env.DB_USERNAME,
    password: process.env.DB_PASSWORD,
    database: process.env.DB_DATABASE,
    // Storage entities are required when using DatabaseStorageAdapter.
    // If you use RedisStorageAdapter instead, you can omit getNAuthTransientStorageEntities().
    entities: [...getNAuthEntities(), ...getNAuthTransientStorageEntities()],
    synchronize: process.env.NODE_ENV === 'development',
  });
  await dataSource.initialize();

  // Create NAuth instance
  const nauth = await NAuth.create({
    config: authConfig,
    dataSource,
    adapter: new FastifyAdapter(),
  });

  const fastify = Fastify();
  await fastify.register(cookie);

  // Register hooks (order matters)
  fastify.addHook('onRequest', nauth.middleware.clientInfo as any); // FIRST
  fastify.addHook('onRequest', nauth.middleware.csrf as any);
  fastify.addHook('onRequest', nauth.middleware.auth as any);
  fastify.addHook('onSend', nauth.middleware.tokenDelivery as any);

  // Routes
  await fastify.register(async (api) => {
    createAuthRoutes(api, nauth);
  }, { prefix: '/auth' });

  await fastify.listen({ port: 3000 });
  console.log('Server running on :3000');
}

bootstrap();
```

```typescript title="routes/auth.routes.ts"
import { FastifyInstance } from 'fastify';
import { NAuthInstance } from '@nauth-toolkit/core';

export async function createAuthRoutes(
  fastify: FastifyInstance,
  nauth: NAuthInstance<any, any>,
): Promise<void> {
  fastify.post(
    '/signup',
    { preHandler: nauth.helpers.public() as any },
    nauth.adapter.wrapRouteHandler(async (req) => {
      return nauth.authService.signup(req.body as any);
    }),
  );

  fastify.post(
    '/login',
    { preHandler: nauth.helpers.public() as any },
    nauth.adapter.wrapRouteHandler(async (req) => {
      return nauth.authService.login(req.body as any);
    }),
  );

  fastify.get(
    '/me',
    { preHandler: nauth.helpers.requireAuth() as any },
    nauth.adapter.wrapRouteHandler(async () => nauth.helpers.getCurrentUser()),
  );
}
```

```typescript title="config/auth.config.ts"
import { NAuthModuleConfig, MFAMethod, createRedisStorageAdapter } from '@nauth-toolkit/nestjs';
import { ConsoleEmailProvider } from '@nauth-toolkit/email-console';
import { ConsoleSMSProvider } from '@nauth-toolkit/sms-console';
import { Logger } from '@nestjs/common';

// For production, use real providers:
// import { NodemailerEmailProvider } from '@nauth-toolkit/email-nodemailer';
// import { AWSSMSProvider } from '@nauth-toolkit/sms-aws-sns';

export const authConfig: NAuthModuleConfig = {
  tablePrefix: 'nauth_',

  // Redis for production (rate limits, locks, token reuse detection)
  storageAdapter: createRedisStorageAdapter(process.env.REDIS_URL!),

  jwt: {
    algorithm: 'HS256',
    issuer: 'com.myapp',
    audience: ['web', 'mobile'],
    accessToken: {
      secret: process.env.JWT_SECRET!,
      expiresIn: '15m',
    },
    refreshToken: {
      secret: process.env.JWT_REFRESH_SECRET!,
      expiresIn: '7d',
      rotation: true,
      reuseDetection: true,
    },
  },

  // Logger configuration
  logger: {
    instance: new Logger('NAuth'),
    enablePiiRedaction: true,
    logLevel: 'info',
  },

  // Signup with detailed verification settings
  signup: {
    enabled: true,
    verificationMethod: 'both', // 'email', 'phone', or 'both'
    allowDuplicatePhones: false,
    emailVerification: {
      expiresIn: 3600, // 1 hour
      resendDelay: 60, // 1 minute between resends
      rateLimitMax: 5, // Max emails per window
      rateLimitWindow: 3600, // 1 hour window
      maxAttemptsPerIP: 10, // Max verification attempts per IP
      attemptWindow: 3600, // 1 hour window for attempts
    },
    phoneVerification: {
      codeLength: 6,
      expiresIn: 300, // 5 minutes
      maxAttempts: 3, // Max verification attempts
      resendDelay: 60, // 1 minute between resends
      rateLimitMax: 5, // Max SMS per window
      rateLimitWindow: 3600, // 1 hour window
      maxAttemptsPerUser: 5, // Max attempts per user
      maxAttemptsPerIP: 10, // Max attempts per IP
      attemptWindow: 3600, // 1 hour window
    },
  },

  // MFA with comprehensive settings
  mfa: {
    enabled: true,
    enforcement: 'ADAPTIVE', // 'OPTIONAL', 'REQUIRED', or 'ADAPTIVE'
    gracePeriod: 0, // Days before MFA becomes required (when enforcement is 'REQUIRED')
    requireForSocialLogin: false,
    allowedMethods: [MFAMethod.SMS, MFAMethod.EMAIL, MFAMethod.TOTP, MFAMethod.PASSKEY],
    issuer: 'My App',
    totp: {
      window: 1, // Accept codes from previous/next time step
      stepSeconds: 30, // Standard 30 seconds (compatible with Google Authenticator, Authy)
      digits: 6,
      algorithm: 'sha1',
    },
    passkey: {
      rpName: 'My App',
      rpId: 'myapp.com', // Relying Party ID (domain)
      origin: ['https://myapp.com', 'https://www.myapp.com'], // Allowed origins
      timeout: 60000, // 60 seconds
      userVerification: 'preferred', // 'required', 'preferred', or 'discouraged'
      authenticatorAttachment: undefined, // 'platform' or 'cross-platform'
    },
    adaptive: {
      triggers: ['new_device', 'new_ip', 'new_country', 'impossible_travel'],
      riskLevels: {
        low: { maxScore: 20, action: 'allow', notifyUser: false },
        medium: { maxScore: 50, action: 'require_mfa', notifyUser: true },
        high: { maxScore: 100, action: 'block_signin', notifyUser: true },
      },
    },
    rememberDevices: 'user_opt_in', // 'never', 'user_opt_in', or 'always'
    rememberDeviceDays: 30,
    bypassMFAForTrustedDevices: true, // Does not apply to Adaptive MFA
  },

  // Password policy with reset settings
  password: {
    minLength: 8,
    requireUppercase: true,
    requireNumbers: true,
    requireSpecialChars: true,
    preventCommon: true,
    historyCount: 5, // Prevent reusing last 5 passwords
    expiryDays: 0, // 0 = never expire, or set days (e.g., 90)
    specialChars: '$#!@', // Allowed special characters
    passwordReset: {
      codeLength: 6,
      expiresIn: 900, // 15 minutes
      rateLimitMax: 3, // Max reset requests per window
      rateLimitWindow: 3600, // 1 hour window
      maxAttempts: 3, // Max verification attempts
    },
  },

  // Token delivery configuration
  tokenDelivery: {
    method: 'hybrid', // 'json', 'cookies', or 'hybrid'
    cookieOptions: {
      secure: true, // HTTPS only
      sameSite: 'strict', // 'strict', 'lax', or 'none'
      domain: '.myapp.com', // Cookie domain (optional)
    },
  },

  // Security settings
  security: {
    csrf: {
      cookieName: 'nauth_csrf_token',
      headerName: 'x-csrf-token',
      cookieOptions: {
        domain: '.myapp.com', // Optional: set cookie domain
      },
    },
  },

  // IP geolocation (MaxMind)
  geoLocation: {
    maxMind: {
      licenseKey: process.env.MAXMIND_LICENSE_KEY,
      accountId: parseInt(process.env.MAXMIND_ACCOUNT_ID || '0', 10),
      autoDownloadOnStartup: false, // Set to true to auto-download on server start
      editions: ['GeoLite2-City', 'GeoLite2-Country'],
    },
```

## Adding Features

The core packages excludes optional features to keep the toolkit lightweight for features that may not be required. However, you can add these features by following the guides on Social, MFA.

### Social Login

```bash npm2yarn
npm install @nauth-toolkit/social-google
```
