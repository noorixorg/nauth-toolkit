## Basic Setup

<Tabs groupId="framework">
<TabItem value="nestjs" label="NestJS" default>

```typescript title="app.module.ts"
import { Module } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';
import { AuthModule } from '@nauth-toolkit/nestjs';
import { getNAuthEntities } from '@nauth-toolkit/database-typeorm-postgres';
import { authConfig } from './config/auth.config';

@Module({
  imports: [
    TypeOrmModule.forRoot({
      type: 'postgres',
      host: process.env.DB_HOST,
      port: parseInt(process.env.DB_PORT || '5432'),
      username: process.env.DB_USERNAME,
      password: process.env.DB_PASSWORD,
      database: process.env.DB_DATABASE,
      entities: getNAuthEntities(),
      synchronize: process.env.NODE_ENV === 'development',
    }),
    AuthModule.forRoot(authConfig),
  ],
})
export class AppModule {}
```

```typescript title="auth.controller.ts"
import { Controller, Post, Get, Body, UseGuards } from '@nestjs/common';
import { AuthService, SignupDTO, LoginDTO, AuthGuard, CurrentUser, Public, IUser } from '@nauth-toolkit/nestjs';

@Controller('auth')
export class AuthController {
  constructor(private readonly authService: AuthService) {}

  @Public()
  @Post('signup')
  async signup(@Body() dto: SignupDTO) {
    return this.authService.signup(dto);
  }

  @Public()
  @Post('login')
  async login(@Body() dto: LoginDTO) {
    return this.authService.login(dto);
  }

  @UseGuards(AuthGuard)
  @Get('me')
  async getCurrentUser(@CurrentUser() user: IUser) {
    return user;
  }
}
```

<TabItem value="express" label="Express">

```typescript title="index.ts"
import express from 'express';
import cookieParser from 'cookie-parser';
import { DataSource } from 'typeorm';
import { NAuth, ExpressAdapter } from '@nauth-toolkit/core';
import { getNAuthEntities } from '@nauth-toolkit/database-typeorm-postgres';
import { authConfig } from './config/auth.config';

async function bootstrap() {
  // Initialize database
  const dataSource = new DataSource({
    type: 'postgres',
    host: process.env.DB_HOST,
    port: parseInt(process.env.DB_PORT || '5432'),
    username: process.env.DB_USERNAME,
    password: process.env.DB_PASSWORD,
    database: process.env.DB_DATABASE,
    entities: getNAuthEntities(),
    synchronize: process.env.NODE_ENV === 'development',
  });
  await dataSource.initialize();

  // Create NAuth instance
  const nauth = await NAuth.create({
    config: authConfig,
    dataSource,
    adapter: new ExpressAdapter(),
  });

  const app = express();
  app.use(express.json());
  app.use(cookieParser());

  // Mount middleware (order matters)
  app.use(nauth.middleware.clientInfo); // FIRST - initializes context
  app.use(nauth.middleware.csrf);
  app.use(nauth.middleware.auth);
  app.use(nauth.middleware.tokenDelivery);

  // Routes
  app.post('/auth/signup', nauth.helpers.public(), async (req, res, next) => {
    try {
      const result = await nauth.authService.signup(req.body);
      res.status(201).json(result);
    } catch (error) {
      next(error);
    }
  });

  app.post('/auth/login', nauth.helpers.public(), async (req, res, next) => {
    try {
      const result = await nauth.authService.login(req.body);
      res.json(result);
    } catch (error) {
      next(error);
    }
  });

  app.get('/auth/me', nauth.helpers.requireAuth(), (req, res) => {
    const user = nauth.helpers.getCurrentUser();
    res.json(user);
  });

  app.listen(3000, () => console.log('Server running on :3000'));
}

bootstrap();
```

<TabItem value="fastify" label="Fastify">

```typescript title="index.ts"
import Fastify from 'fastify';
import cookie from '@fastify/cookie';
import { DataSource } from 'typeorm';
import { NAuth, FastifyAdapter } from '@nauth-toolkit/core';
import { getNAuthEntities } from '@nauth-toolkit/database-typeorm-postgres';
import { authConfig } from './config/auth.config';

async function bootstrap() {
  // Initialize database
  const dataSource = new DataSource({
    type: 'postgres',
    host: process.env.DB_HOST,
    port: parseInt(process.env.DB_PORT || '5432'),
    username: process.env.DB_USERNAME,
    password: process.env.DB_PASSWORD,
    database: process.env.DB_DATABASE,
    entities: getNAuthEntities(),
    synchronize: process.env.NODE_ENV === 'development',
  });
  await dataSource.initialize();

  // Create NAuth instance
  const nauth = await NAuth.create({
    config: authConfig,
    dataSource,
    adapter: new FastifyAdapter(),
  });

  const fastify = Fastify();
  await fastify.register(cookie);

  // Register hooks (order matters)
  fastify.addHook('onRequest', nauth.middleware.clientInfo as any); // FIRST
  fastify.addHook('onRequest', nauth.middleware.csrf as any);
  fastify.addHook('onRequest', nauth.middleware.auth as any);
  fastify.addHook('onSend', nauth.middleware.tokenDelivery as any);

  // Routes - Fastify handlers must be wrapped for context propagation
  fastify.post(
    '/auth/signup',
    { preHandler: nauth.helpers.public() as any },
    nauth.adapter.wrapRouteHandler(async (req) => {
      return nauth.authService.signup(req.body as any);
    }),
  );

  fastify.post(
    '/auth/login',
    { preHandler: nauth.helpers.public() as any },
    nauth.adapter.wrapRouteHandler(async (req) => {
      return nauth.authService.login(req.body as any);
    }),
  );

  fastify.get(
    '/auth/me',
    { preHandler: nauth.helpers.requireAuth() as any },
    nauth.adapter.wrapRouteHandler(async () => nauth.helpers.getCurrentUser()),
  );

  await fastify.listen({ port: 3000 });
  console.log('Server running on :3000');
}

bootstrap();
```

import { NAuthConfig } from '@nauth-toolkit/core';

export const authConfig: NAuthConfig = {

    },
    refreshToken: {
      secret: process.env.JWT_REFRESH_SECRET!,
      expiresIn: '7d',
    },

  password: {
    minLength: 8,
    requireUppercase: true,
    requireNumbers: true,
    requireSpecialChars: true,

      host: process.env.SMTP_HOST,
      port: parseInt(process.env.SMTP_PORT || '587'),
      auth: {
        user: process.env.SMTP_USER,
        pass: process.env.SMTP_PASS,
      },
    },
    defaults: { from: 'noreply@myapp.com' },
  }),

  signup: {
    enabled: true,
    verificationMethod: 'email',
  },
};

## Key Differences by Framework

| Aspect | NestJS | Express | Fastify |
|--------|--------|---------|---------|
| **Bootstrap** | `AuthModule.forRoot()` | `NAuth.create()` | `NAuth.create()` |
| **Adapter** | Automatic | `ExpressAdapter` | `FastifyAdapter` |
| **Route protection** | `@UseGuards(AuthGuard)` | `nauth.helpers.requireAuth()` | `preHandler` + `nauth.helpers.requireAuth()` |
| **Public routes** | `@Public()` decorator | `nauth.helpers.public()` | `nauth.helpers.public()` |
| **Get current user** | `@CurrentUser()` decorator | `nauth.helpers.getCurrentUser()` | `nauth.helpers.getCurrentUser()` |
| **Context propagation** | Automatic | Automatic | Use `adapter.wrapRouteHandler` on route handlers |
